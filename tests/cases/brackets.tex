%% \ExLBr and its family are linguex's names for the characters around an
%% example number, and a document is entitled to redefine them --
%% \renewcommand{\ExLBr}{[} is the documented way to get [1] instead of (1).
%% linguexx has honoured that since 1.3.
%%
%% This converter emitted parentheses regardless, with no warning: a wrong
%% page that still compiles, which is the failure mode linguexx's own 1.3
%% changelog calls the one a drop-in replacement cannot have.  Nothing in
%% the document body is unknown, so the pandoc fallback could never catch
%% it; only a preamble scan can.
%%
%% \SubExRBr is here too because the sub-example markers come through the
%% same mechanism -- 'a.' is SubExLBr + letter + SubExRBr, with the period
%% supplied by the default -- and a label must still point at the bare
%% letter once the marker around it can be something else.
\input{_preamble}
\renewcommand{\ExLBr}{[}
\renewcommand{\ExRBr}{]}
\renewcommand{\SubExRBr}{)}
\begin{document}
\ex.\label{top} Square brackets around the number.
\a. sub one
\b.\label{sub} sub two, whose letter is closed by a parenthesis

Refs: \ref{top}, \pref{top}, \ref{sub}.
\end{document}
