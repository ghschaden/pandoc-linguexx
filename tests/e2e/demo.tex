\section{Glossed examples}

\ex.\label{ex:span} \gll Esto es un ejemplo glosado \\
     this is a example glossed \\
\glt `This is a glossed example.'

\ex. \gll Das {kleine Kind} schl\"aft \\
     the {little child} sleeps \\
\glt `The little child sleeps.'

\ex. \glll Ich habe geschlafen \\
     ich hab-e schlaf-en \\
     \lpzg{1sg} have-\lpzg{1sg} sleep-\lpzg{ptcp} \\
\glt `I slept.'
